\subsection*{Setup}


\begin{itemize}

\item Chain length $n=25$, closed chain, $\gamma=0.1$, $dt=0.001$,
$T_{\mathrm{final}}=200000$, $T_{\mathrm{burnin}}=2000$
\item $32$ groups of $W=16$ SIMD lanes, so $512$ independent chains
\item for every step, accumulate into the histogram
only on lanes for which $|\sum_j I_j - S| < \delta$, with $S = n/2 = 12.5$
and $\delta = 0.01\,n = 0.25$
\item Sites: $j=6, 12, 18$ 
\item $1.01\times 10^{11}$ steps per chain set.
accepted samples: $7.75\times 10^9$ (NESS, $7.6\%$ acceptance),
$1.10\times 10^{10}$ (equilibrium, $10.8\%$ acceptance)

\end{itemize}

\subsection*{CI}
For each bin $b$ in the masked fit region (counts $\geq 50$, $I<3.2$),
the Agresti Coull 95\% interval on the bin proportion $p_b=n_b/N$ is
\[
\tilde p_b = \frac{n_b+2}{N+4}, \qquad
\text{halfwidth}(p_b) = z\sqrt{\tilde p_b(1-\tilde p_b)/(N+4)}, \quad z=1.96.
\]
The 1-$\sigma$ standard error on $\log\rho_b$ is taken as
$\sigma_b = \text{halfwidth}(p_b)/(z\,\hat p_b)$ 
\begin{align*}
\text{Form A (conjecture)} \quad
& \log\rho = a_0 + a_1\!\left[-\tfrac{1}{4}(I_a^2+I_b^2) + I_a I_b\cos\theta\right] \\
\text{Form B (strict local Gibbs)} \quad
& \log\rho = b_0 + b_1\!\left[\tfrac{1}{4}(I_a^2+I_b^2) + I_a I_b(1+\cos\theta)\right] \\
\text{Form C (general quadratic)} \quad
& \log\rho = c_0 + c_1 I_a^2 + c_2 I_b^2 + c_3 I_a I_b + c_4 I_a I_b\cos\theta
\end{align*}
are fit by weighted least squares with weights $1/\sigma_b^2$.
Coefficient covariances come from $(X^\top W X)^{-1}$; ratios and their
errors come from the delta method
