\section*{Setup}

\begin{itemize}
    \item closed chain, $T_1=10$, $T_n=2$, $\gamma=0.1$
    \item local energy flux across $m=\lfloor n/2\rfloor$: $J=4\,I_{m1}\,I_m\,\sin\!\big(2(\varphi_m-\varphi_{m-1})\big)$.

    \item $dt=5\times10^{-4}$, burn-in $1000$, measurement $\tau=200$.  For each $n\in\{10,20,30,40\}$, collect $10^4$ trajectories.

\end{itemize}

\begin{itemize}
    \item The mean flux drops from $0.42$ at $n=10$ to $0.029$ at $n=40$, scaling as
$\langle J\rangle\sim n^{-1.91}$. The corresponding conductivity
$\kappa\sim \langle J\rangle\,n\sim n^{-0.9}\to0$

\item The tail looks exponential. Fit
$\log P[J>x]=-\lambda x+c$ over $[5\times10^{-3},\,0.2]$ gives $R^2\ge0.993$ for every $n$
\end{itemize}
